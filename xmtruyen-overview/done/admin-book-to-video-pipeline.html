1. Tổng Quan — Sách Chữ Khác Truyện Tranh Ở Đâu?

🖼️ Truyện Tranh → Video (đã có tài liệu)

  • Input: ảnh comic pages có sẵn
  • Mỗi ảnh = 1 khung hình video
  • TTS đọc lời thoại/narration
  • Đơn giản: ảnh → slideshow + audio

📖 Sách Chữ → Video (tài liệu này)

  • Input: chỉ có text, KHÔNG có ảnh
  • Cần tạo hình minh họa cho nội dung
  • TTS đọc toàn bộ nội dung truyện chữ
  • Phức tạp: text → AI images → audio → video
💡 Ý tưởng: Giống các kênh YouTube "Đọc truyện" — video hiển thị hình minh họa đẹp (cảnh, nhân vật) trong khi giọng AI kể chuyện. User xem video thay vì đọc text → tiếp cận nhiều đối tượng hơn (lái xe, nấu ăn, trước khi ngủ...).

3.2 AI Scene Description Generator

📁 xmtruyen.API/Services/SceneDescriptionService.cs
/// <summary>
/// Phân tích text truyện → tạo mô tả cảnh (scene description) cho mỗi đoạn.
/// Scene description dùng làm prompt cho AI image generation.
/// </summary>
public class SceneDescriptionService
{
    /// <summary>
    /// Chia text thành segments + tạo scene description cho mỗi segment.
    /// Dùng GPT / local LLM để phân tích context.
    /// </summary>
    public async Task<List<TextSegment>> AnalyzeAndSplitAsync(string chapterText, string genre, string style)
    {
        // 1. Chia text thành đoạn ~150 từ theo paragraph breaks
        var rawSegments = SplitByParagraphs(chapterText, targetWords: 150);
        
        var results = new List<TextSegment>();
        
        foreach (var segment in rawSegments)
        {
            // 2. Tạo scene description bằng AI (hoặc rule-based)
            var scenePrompt = await GenerateScenePromptAsync(segment, genre, style);
            
            results.Add(new TextSegment
            {
                Text = segment,
                SceneDescription = scenePrompt,
                EstimatedDurationSeconds = EstimateReadingDuration(segment),
                Keywords = ExtractKeywords(segment)
            });
        }
        
        return results;
    }

    /// <summary>
    /// Tạo prompt cho AI image generation từ nội dung đoạn văn.
    /// </summary>
    private async Task<string> GenerateScenePromptAsync(string text, string genre, string style)
    {
        // Option A: Dùng GPT API tạo prompt
        // "Mô tả cảnh trong 1 câu tiếng Anh, phong cách {style}, thể loại {genre}"
        
        // Option B: Rule-based (không cần AI)
        // Phân tích: tìm danh từ chỉ nơi chốn, nhân vật, hành động
        var keywords = ExtractKeywords(text);
        var prompt = $"{style} illustration, {genre} genre, scene: {string.Join(", ", keywords)}, " +
                     $"cinematic lighting, detailed background, 16:9 aspect ratio";
        
        return prompt;
    }

    /// <summary>
    /// Phong cách ảnh tùy theo thể loại truyện
    /// </summary>
    public static string GetStyleByGenre(string genre) => genre switch
    {
        "Tiên Hiệp" => "chinese fantasy art, cultivation immortal, ethereal mountains, martial arts",
        "Kiếm Hiệp" => "ancient chinese warrior, sword fight, misty mountains, wuxia style",
        "Ngôn Tình" => "romantic illustration, soft lighting, modern couple, pastel colors",
        "Đô Thị" => "modern city, urban lifestyle, photorealistic, contemporary",
        "Huyền Huyễn" => "dark fantasy, magical world, epic landscape, dramatic lighting",
        "Trinh Thám" => "noir detective, dark atmosphere, mystery, shadowy",
        "Sci-Fi" => "science fiction, futuristic city, neon lights, cyberpunk",
        _ => "digital art, illustration, cinematic, detailed"
    };
}

3.3 Image Generation Service

📁 xmtruyen.API/Services/ImageGenerationService.cs
public class ImageGenerationService : IImageGenerationService
{
    private readonly HttpClient _httpClient;
    private readonly IConfiguration _config;

    /// <summary>
    /// Tạo ảnh minh họa bằng AI (hỗ trợ nhiều provider)
    /// </summary>
    public async Task<string> GenerateImageAsync(string prompt, string outputPath, string provider = "stable-diffusion")
    {
        return provider switch
        {
            "dall-e" => await GenerateWithDallE(prompt, outputPath),
            "stable-diffusion" => await GenerateWithStableDiffusion(prompt, outputPath),
            "stock" => await FetchStockImage(prompt, outputPath),
            _ => throw new ArgumentException($"Unknown provider: {provider}")
        };
    }

    /// <summary> DALL-E 3 (OpenAI) — $0.04/ảnh, chất lượng cao </summary>
    private async Task<string> GenerateWithDallE(string prompt, string outputPath)
    {
        var apiKey = _config["OpenAI:ApiKey"];
        var request = new
        {
            model = "dall-e-3",
            prompt = prompt,
            n = 1,
            size = "1792x1024",  // 16:9 ratio cho video
            quality = "standard"
        };

        var response = await _httpClient.PostAsJsonAsync(
            "https://api.openai.com/v1/images/generations", request,
            new JsonSerializerOptions { });

        // Download image URL → save to outputPath
        var imageUrl = /* parse response */;
        await DownloadFileAsync(imageUrl, outputPath);
        return outputPath;
    }

    /// <summary> Stable Diffusion (Local hoặc API) — Miễn phí nếu chạy local </summary>
    private async Task<string> GenerateWithStableDiffusion(string prompt, string outputPath)
    {
        // Gọi Stable Diffusion WebUI API (chạy local trên GPU):
        // POST http://127.0.0.1:7860/sdapi/v1/txt2img
        var sdRequest = new
        {
            prompt = prompt,
            negative_prompt = "blurry, low quality, text, watermark, distorted",
            width = 1280,
            height = 720,
            steps = 25,
            cfg_scale = 7.5,
            sampler_name = "DPM++ 2M Karras"
        };

        var response = await _httpClient.PostAsJsonAsync(
            "http://127.0.0.1:7860/sdapi/v1/txt2img", sdRequest);

        // Response chứa base64 image → decode + save
        var result = await response.Content.ReadFromJsonAsync<SdResponse>();
        var imageBytes = Convert.FromBase64String(result!.Images[0]);
        await File.WriteAllBytesAsync(outputPath, imageBytes);
        return outputPath;
    }

    /// <summary> Stock Images (Unsplash/Pexels) — Miễn phí, fallback </summary>
    private async Task<string> FetchStockImage(string keywords, string outputPath)
    {
        var apiKey = _config["Unsplash:AccessKey"];
        var searchUrl = $"https://api.unsplash.com/search/photos?query={Uri.EscapeDataString(keywords)}&orientation=landscape&per_page=1";
        
        _httpClient.DefaultRequestHeaders.Add("Authorization", $"Client-ID {apiKey}");
        var response = await _httpClient.GetFromJsonAsync<UnsplashResponse>(searchUrl);
        
        var imageUrl = response?.Results?.FirstOrDefault()?.Urls?.Regular;
        if (string.IsNullOrEmpty(imageUrl)) 
            throw new Exception("Không tìm thấy ảnh phù hợp");

        await DownloadFileAsync(imageUrl, outputPath);
        return outputPath;
    }
}

4. TTS — Đọc Nội Dung Truyện (Đa Ngôn Ngữ)

Khác biệt với truyện tranh: Truyện chữ có text dài (2000-5000 từ/chương). TTS cần đọc TOÀN BỘ nội dung — không chỉ narration ngắn. Cần chia thành segments để sync với ảnh.

4.1 Luồng xử lý TTS cho sách chữ

Chương text (2500 từ) │ ▼ Chia thành ~15 segments (~150 từ/segment) │ ├──▶ Segment 1: "Lâm Động đứng trên đỉnh núi..." (32s audio) ├──▶ Segment 2: "Phía xa, một bóng người xuất hiện..." (28s audio) ├──▶ Segment 3: "Cô gái mặc áo trắng bước tới..." (35s audio) ├──▶ ... └──▶ Segment 15: "Mặt trời lặn dần phía chân trời..." (30s audio) Tổng audio: ~7-8 phút cho 1 chương Tổng video: ~7-8 phút (15 scenes × ~30s mỗi scene)

4.3 Tính năng TTS nâng cao

  • SSML markup: Thêm ngắt nghỉ (pause) giữa các đoạn hội thoại, nhấn mạnh từ quan trọng
  • Multi-voice: Giọng khác nhau cho dialogue (narrator = voice A, nhân vật nữ = voice B, nhân vật nam = voice C)
  • Emotion tags: Thay đổi giọng đọc theo cảm xúc đoạn văn (vui, buồn, giận, sợ)
  • Background SFX: Auto-thêm âm thanh nền phù hợp (tiếng gió, tiếng kiếm, tiếng mưa) dựa trên keywords
📁 Ví dụ SSML cho Edge TTS
<speak version="1.0" xmlns="http://www.w3.org/2001/10/synthesis" xml:lang="vi-VN">
    <voice name="vi-VN-NamMinhNeural">
        <prosody rate="-5%" pitch="-2st">
            Lâm Động đứng trên đỉnh núi, nhìn xa xăm.
        </prosody>
        <break time="500ms"/>
        <prosody rate="+0%" pitch="+0st" volume="+10%">
            "Ngươi là ai?!" — Hắn quát lớn.
        </prosody>
        <break time="300ms"/>
    </voice>
</speak>

5.1 FFmpeg Compose Service

📁 xmtruyen.API/Services/BookVideoComposeService.cs
public class BookVideoComposeService
{
    /// <summary>
    /// Ghép video storytelling từ segments (image + audio + subtitle)
    /// </summary>
    public async Task<string> ComposeBookVideoAsync(BookVideoRequest request)
    {
        var segments = request.Segments;
        var outputDir = request.WorkingDir;
        var segmentFiles = new List<string>();

        // 1. Tạo Intro segment
        var introPath = await CreateIntroSegment(request.BookTitle, request.ChapterTitle, 
            request.AuthorName, request.CoverImagePath, outputDir);
        segmentFiles.Add(introPath);

        // 2. Tạo từng scene segment
        for (int i = 0; i < segments.Count; i++)
        {
            var seg = segments[i];
            var segPath = Path.Combine(outputDir, $"scene_{i:D3}.mp4");
            
            // Chọn hiệu ứng random (tránh monotone)
            var effect = GetRandomEffect(i);
            
            // FFmpeg: ảnh + audio + subtitle overlay → segment video
            var ffArgs = $"-loop 1 -i \"{seg.ImagePath}\" " +
                         $"-i \"{seg.AudioPath}\" " +
                         $"-filter_complex \"" +
                         // Scale + pad to 1080p
                         $"[0:v]scale=1920:1080:force_original_aspect_ratio=decrease," +
                         $"pad=1920:1080:(ow-iw)/2:(oh-ih)/2," +
                         // Ken Burns / Pan effect
                         $"{effect}," +
                         $"format=yuv420p[v];" +
                         // Subtitle overlay (nếu có)
                         (request.AddSubtitles ? 
                             $"[v]drawtext=text='{EscapeText(seg.SubtitleText)}':" +
                             $"fontcolor=white:fontsize=28:borderw=2:bordercolor=black:" +
                             $"x=(w-text_w)/2:y=h-th-60[final]\" " :
                             $"[v]null[final]\" ") +
                         $"-map \"[final]\" -map 1:a " +
                         $"-c:v libx264 -preset fast -crf 23 -c:a aac -b:a 192k " +
                         $"-t {seg.AudioDuration:F2} -shortest \"{segPath}\"";
            
            await RunFFmpegAsync(ffArgs);
            segmentFiles.Add(segPath);
        }

        // 3. Tạo Outro segment
        var outroPath = await CreateOutroSegment(request.BookTitle, outputDir);
        segmentFiles.Add(outroPath);

        // 4. Concat tất cả segments
        var concatFile = Path.Combine(outputDir, "concat.txt");
        var concatLines = segmentFiles.Select(f => $"file '{f}'").ToList();
        await File.WriteAllLinesAsync(concatFile, concatLines);
        
        var finalOutput = Path.Combine(outputDir, "final_output.mp4");
        await RunFFmpegAsync($"-f concat -safe 0 -i \"{concatFile}\" -c copy \"{finalOutput}\"");

        // 5. Mix nhạc nền (nếu có)
        if (!string.IsNullOrEmpty(request.BackgroundMusicPath))
        {
            var withBgm = Path.Combine(outputDir, "final_bgm.mp4");
            await RunFFmpegAsync(
                $"-i \"{finalOutput}\" -i \"{request.BackgroundMusicPath}\" " +
                $"-filter_complex \"[1:a]volume={request.BgmVolume}[bgm];" +
                $"[0:a][bgm]amix=inputs=2:duration=first[aout]\" " +
                $"-map 0:v -map \"[aout]\" -c:v copy -c:a aac \"{withBgm}\"");
            File.Delete(finalOutput);
            File.Move(withBgm, finalOutput);
        }

        return finalOutput;
    }

    private string GetRandomEffect(int index)
    {
        var effects = new[]
        {
            "zoompan=z='min(zoom+0.001,1.08)':d=750:x='iw/2-(iw/zoom/2)':y='ih/2-(ih/zoom/2)':s=1920x1080",  // Zoom in
            "zoompan=z='1.08':d=750:x='(iw-iw/zoom)/2+((iw/zoom)*0.01*on/750)':y='ih/2-(ih/zoom/2)':s=1920x1080",  // Pan right
            "zoompan=z='if(eq(on,1),1.08,zoom-0.0001)':d=750:x='iw/2-(iw/zoom/2)':y='ih/2-(ih/zoom/2)':s=1920x1080",  // Zoom out
        };
        return effects[index % effects.Length];
    }
}

8. Background Worker

📁 Luồng xử lý trong Worker (progress tracking)
// VideoConvertWorker steps cho Book → Video:

Step 1: [  0-5%  ] Fetch chapter text từ DB
Step 2: [  5-15% ] Phân tích text → segments (SceneDescriptionService)
Step 3: [ 15-50% ] Tạo ảnh minh họa cho mỗi segment (AI/Stock)
         → Bước này lâu nhất nếu dùng AI (~15s/ảnh × 15 segments = 3-4 phút)
Step 4: [ 50-70% ] TTS → generate audio cho mỗi segment
Step 5: [ 70-90% ] FFmpeg → ghép video (intro + scenes + outro + BGM)
Step 6: [ 90-95% ] Generate subtitle file (.srt)
Step 7: [ 95-100%] Upload output + update DB status = "Completed"

// Admin thấy progress bar real-time + step description

CREATE TABLE "BookVideoTasks" (

    "Resolution" VARCHAR(10) DEFAULT '1080p',
    "Transition" VARCHAR(20) DEFAULT 'kenburns',
    "AddSubtitles" BOOLEAN DEFAULT TRUE,
    "AddIntroOutro" BOOLEAN DEFAULT TRUE,
    "BgmEnabled" BOOLEAN DEFAULT FALSE,
    "BgmGenre" VARCHAR(50),
    "BgmVolume" FLOAT DEFAULT 0.2,

    -- Progress
    "Status" VARCHAR(20) DEFAULT 'Queued',
    "ProgressPercent" INT DEFAULT 0,
    "CurrentStep" TEXT,

    "TotalSegments" INT DEFAULT 0,
    "CompletedSegments" INT DEFAULT 0,

    -- Output

Video có audio → user extract audio hoặc bật video background → nghe truyện khi không thể đọc text.
