3.1 Code: MagazineFlip Component

📁 xom-truyen/src/components/Reading/Effects/MagazineFlip.tsx
import React, { useRef, useCallback } from 'react';
import HTMLFlipBook from 'react-pageflip';

interface MagazineFlipProps {
    pages: { content: string; type: 'text' | 'image' }[];
    currentPage: number;
    onPageChange: (page: number) => void;
    width?: number;
    height?: number;
    showCover?: boolean;
}

/**
 * Magazine-style page flip sử dụng react-pageflip (StPageFlip engine).
 * Effect: 3D realistic page turn với shadow, fold, corner drag.
 */
export const MagazineFlip: React.FC<MagazineFlipProps> = ({
    pages, currentPage, onPageChange, width = 600, height = 800, showCover = true
}) => {
    const flipBookRef = useRef<any>(null);

    const handleFlip = useCallback((e: any) => {
        onPageChange(e.data); // page index
    }, [onPageChange]);

    return (
        <div className="magazine-flip-container" style={{
            display: 'flex', justifyContent: 'center', alignItems: 'center',
            perspective: '2000px', minHeight: '80vh'
        }}>
            <HTMLFlipBook
                ref={flipBookRef}
                width={width}
                height={height}
                size="stretch"
                minWidth={300}
                maxWidth={1000}
                minHeight={400}
                maxHeight={1200}
                showCover={showCover}
                mobileScrollSupport={true}
                onFlip={handleFlip}
                className="magazine-flipbook"
                style={{}}
                startPage={currentPage}
                drawShadow={true}
                flippingTime={800}
                usePortrait={window.innerWidth < 768}
                startZIndex={0}
                autoSize={true}
                maxShadowOpacity={0.5}
                showPageCorners={true}
                disableFlipByClick={false}
            >
                {pages.map((page, idx) => (
                    <div key={idx} className="flip-page" data-density={idx === 0 ? 'hard' : 'soft'}>
                        {page.type === 'image' ? (
                            <img src={page.content} alt={`Page ${idx + 1}`}
                                 style={{ width: '100%', height: '100%', objectFit: 'contain' }} />
                        ) : (
                            <div className="text-page-content" style={{
                                padding: '40px 50px', fontSize: '16px', lineHeight: 1.8,
                                fontFamily: 'Literata, Georgia, serif',
                                backgroundColor: '#fefcf7', // Giấy sách hơi vàng
                                height: '100%', overflow: 'hidden'
                            }}>
                                <div dangerouslySetInnerHTML={{ __html: page.content }} />
                            </div>
                        )}
                    </div>
                ))}
            </HTMLFlipBook>
        </div>
    );
};

5. Effect: Slider (Trượt Ngang Kiểu Presentation)

Mô tả:

Trang trượt ngang smooth (translateX). Không có hiệu ứng lật 3D. Giống xem slides PowerPoint hoặc carousel.

Khi nào dùng:

  • Truyện tranh muốn xem từng trang (không cuộn dọc)
  • Mobile — swipe trái/phải cảm giác tự nhiên
  • Performance tốt nhất (chỉ translateX, GPU accelerated)

Thư viện:

Dùng Swiper.js (đã có trong project) với effect="slide"

📁 xom-truyen/src/components/Reading/Effects/SliderView.tsx
import React from 'react';
import { Swiper, SwiperSlide } from 'swiper/react';
import { Navigation, Keyboard, Pagination } from 'swiper/modules';
import 'swiper/css';

export const SliderView: React.FC<{ pages: PageContent[]; onPageChange: (idx: number) => void }> = 
    ({ pages, onPageChange }) => {
    return (
        <Swiper
            modules={[Navigation, Keyboard, Pagination]}
            navigation
            keyboard={{ enabled: true }}
            pagination={{ type: 'fraction' }}
            onSlideChange={(swiper) => onPageChange(swiper.activeIndex)}
            style={{ width: '100%', height: '80vh' }}
            speed={400}
            spaceBetween={0}
            slidesPerView={1}
        >
            {pages.map((page, idx) => (
                <SwiperSlide key={idx}>
                    {page.type === 'image'
                        ? <img src={page.content} style={{ width: '100%', height: '100%', objectFit: 'contain' }} />
                        : <div className="text-content" dangerouslySetInnerHTML={{ __html: page.content }} />
                    }
                </SwiperSlide>
            ))}
        </Swiper>
    );
};

7. Effect: Notebook (Lật Từ Trên Xuống)

Mô tả:

Trang lật từ trên xuống dưới (hoặc ngược lại), như lật sổ tay xoắn ốc. Transform-origin ở cạnh trên.

CSS core:

.notebook-page.flipping {
    transform-origin: top center;
    animation: notebookFlip 600ms ease forwards;
}
@keyframes notebookFlip {
    from { transform: rotateX(0deg); }
    to { transform: rotateX(-180deg); }
}

Khi nào dùng:

Diary, nhật ký đọc, format content ngắn (1 đoạn/trang).

8. Effect: Cards (Xếp Chồng Bật Ra)

Mô tả:

Các trang xếp chồng lên nhau. Swipe/drag → trang trên "bật ra" (scale down + translate) → trang dưới lộ ra. Giống Tinder swipe.

CSS core:

.card-stack .card:nth-child(1) { transform: scale(1) translateY(0); z-index: 3; }
.card-stack .card:nth-child(2) { transform: scale(0.95) translateY(10px); z-index: 2; opacity: 0.8; }
.card-stack .card:nth-child(3) { transform: scale(0.9) translateY(20px); z-index: 1; opacity: 0.6; }

.card.swiping {
    transition: transform 300ms ease;
    transform: translateX(var(--swipe-x)) rotate(var(--swipe-rotation));
}

Khi nào dùng:

Content ngắn (quotes, flashcards), hoặc browse nhanh nhiều trang.

9. Effect: One Page (Cuộn/Fade Đơn Giản)

Mô tả:

Không có hiệu ứng lật phức tạp. Chuyển trang bằng fade hoặc slide đơn giản. Hoặc cuộn dọc liên tục (mode hiện tại của Xóm Truyện).

Transitions:

/* Fade transition */
.page-enter { opacity: 0; }
.page-enter-active { opacity: 1; transition: opacity 300ms ease; }
.page-exit { opacity: 1; }
.page-exit-active { opacity: 0; transition: opacity 300ms ease; }

/* Slide up transition */
.page-enter { transform: translateY(20px); opacity: 0; }
.page-enter-active { transform: translateY(0); opacity: 1; transition: all 300ms ease; }

Khi nào dùng:

Default fallback, performance tốt nhất, đọc text dài liên tục.

import React, { lazy, Suspense } from 'react';

type PageEffect = 'magazine' | 'book' | 'slider' | 'coverflow' | 'notebook' | 'cards' | 'onepage';

// Lazy load từng effect (code splitting — chỉ load effect đang dùng)
const MagazineFlip = lazy(() => import('./MagazineFlip'));
const BookFlip = lazy(() => import('./BookFlip'));
const SliderView = lazy(() => import('./SliderView'));
const CoverflowView = lazy(() => import('./CoverflowView'));
const NotebookFlip = lazy(() => import('./NotebookFlip'));
const CardsStack = lazy(() => import('./CardsStack'));
const OnePageView = lazy(() => import('./OnePageView'));

interface Props {
    effect: PageEffect;
    pages: { content: string; type: 'text' | 'image' }[];
    currentPage: number;
    onPageChange: (page: number) => void;
}

export const PageEffectWrapper: React.FC<Props> = ({ effect, pages, currentPage, onPageChange }) => {
    const EffectComponent = {
        magazine: MagazineFlip,
        book: BookFlip,
        slider: SliderView,
        coverflow: CoverflowView,
        notebook: NotebookFlip,
        cards: CardsStack,
        onepage: OnePageView,
    }[effect];

    return (
        <Suspense fallback={<div style={{ textAlign: 'center', padding: 40 }}>Đang tải...</div>}>
            <EffectComponent pages={pages} currentPage={currentPage} onPageChange={onPageChange} />
        </Suspense>
    );
};

    // Chia text thành pages ~500 từ mỗi trang
    return splitTextIntoPages(textContent, 500);
}, [textContent]);

// Render:
{!isComicMode ? (
    // Sách chữ → dùng page effect
    <PageEffectWrapper
        effect={pageEffect}
        pages={textPages.map(t => ({ content: t, type: 'text' }))}
        currentPage={currentChapterIndex}
        onPageChange={setCurrentChapterIndex}
    />
) : (
    // Truyện tranh → cuộn dọc (giữ nguyên) hoặc dùng effect
    <PageEffectWrapper
        effect={pageEffect}
        pages={imageUrls.map(url => ({ content: url, type: 'image' }))}
        currentPage={0}
        onPageChange={() => {}}
    />
)}
