1. Tổng Quan & Mục Tiêu

Nhu cầu: Admin cần một nơi tập trung để xem toàn bộ file truyện (ebook, audio, ảnh cover, chapter images...) đang lưu ở đâu, dung lượng bao nhiêu, và có thể switch giữa Server local ↔ Google Drive.

Mục tiêu module:

  • 📊 Dashboard tổng quan: Hiển thị dung lượng đã dùng / còn trống cho mỗi storage (Server, Google Drive)
  • 📁 File Browser: Duyệt thư mục/file trên từng storage — giống File Explorer
  • 🔄 Switch Storage: Click chuyển đổi qua lại giữa Server ↔ Google Drive để xem file
  • 📤 Move/Copy: Di chuyển file từ Server → Google Drive và ngược lại
  • 🗑️ Delete: Xóa file/folder trên cả 2 storage
  • ⚠️ Alert: Cảnh báo khi dung lượng gần đầy (>80%, >90%)
  • 📈 Thống kê: Tổng số file, tổng GB, phân bổ theo loại (ebook, audio, image...)

5. Tích Hợp Google Drive API

5.1 Chuẩn bị

  1. Tạo project trên Google Cloud Console: console.cloud.google.com
  2. Enable Google Drive API trong APIs & Services
  3. Tạo Service Account: IAM & Admin → Service Accounts → Create → Download JSON key
  4. Share folder Google Drive với email service account (ví dụ: xmtruyen-storage@project.iam.gserviceaccount.com)
  5. Lưu JSON key vào server tại path an toàn (ví dụ: /etc/xmtruyen/gdrive-service-account.json)

dotnet add package Google.Apis.Drive.v3

5.3 Google Drive Provider Implementation

📁 Services/Storage/GoogleDriveStorageProvider.cs
using Google.Apis.Auth.OAuth2;
using Google.Apis.Drive.v3;
using Google.Apis.Services;

public class GoogleDriveStorageProvider : IStorageProvider
{
    private readonly DriveService _driveService;
    private readonly string _rootFolderId; // Folder gốc trên Drive

    public string ProviderId => "google-drive";
    public string Label => "Google Drive";

    public GoogleDriveStorageProvider(IConfiguration config)
    {
        var credentialPath = config["Storage:GoogleDrive:CredentialPath"];
        _rootFolderId = config["Storage:GoogleDrive:RootFolderId"];

        var credential = GoogleCredential
            .FromFile(credentialPath)
            .CreateScoped(DriveService.ScopeConstants.Drive);

        _driveService = new DriveService(new BaseClientService.Initializer
        {
            HttpClientInitializer = credential,
            ApplicationName = "Xmtruyen Storage Manager"
        });
    }

    public async Task<StorageOverviewDto> GetOverviewAsync()
    {
        // Lấy quota từ About API
        var aboutRequest = _driveService.About.Get();
        aboutRequest.Fields = "storageQuota";
        var about = await aboutRequest.ExecuteAsync();

        var quota = about.StorageQuota;
        var totalSpace = quota.Limit ?? 16106127360; // 15GB default
        var usedSpace = quota.Usage ?? 0;

        // Đếm files/folders trong root folder
        var (fileCount, folderCount) = await CountItemsRecursive(_rootFolderId);

        return new StorageOverviewDto
        {
            Provider = ProviderId,
            Label = Label,
            TotalSpace = totalSpace,
            UsedSpace = usedSpace,
            FreeSpace = totalSpace - usedSpace,
            TotalFiles = fileCount,
            TotalFolders = folderCount
        };
    }

    public async Task<StorageListResult> ListFilesAsync(string path, string? search, string sortBy, bool isDesc)
    {
        // path = "/" → list root folder
        // path = "/subfolder" → tìm folder ID theo tên → list children
        var folderId = await ResolveFolderIdByPath(path);

        var request = _driveService.Files.List();
        request.Q = $"'{folderId}' in parents and trashed = false";
        if (!string.IsNullOrEmpty(search))
            request.Q += $" and name contains '{search}'";
        request.Fields = "files(id, name, mimeType, size, createdTime, modifiedTime, parents)";
        request.OrderBy = GetOrderBy(sortBy, isDesc);
        request.PageSize = 1000;

        var result = await request.ExecuteAsync();

        var items = result.Files.Select(f => new StorageFileItemDto
        {
            Id = f.Id,
            Name = f.Name,
            Path = path == "/" ? $"/{f.Name}" : $"{path}/{f.Name}",
            Type = f.MimeType == "application/vnd.google-apps.folder" ? "folder" : "file",
            Size = f.Size ?? 0,
            MimeType = f.MimeType,
            CreatedAt = f.CreatedTimeDateTimeOffset?.UtcDateTime ?? DateTime.MinValue,
            ModifiedAt = f.ModifiedTimeDateTimeOffset?.UtcDateTime
        }).ToList();

        return new StorageListResult
        {
            Items = items,
            TotalCount = items.Count,
            CurrentPath = path,
            Breadcrumbs = BuildBreadcrumbs(path)
        };
    }

    public async Task<bool> DeleteAsync(string fileId)
    {
        await _driveService.Files.Delete(fileId).ExecuteAsync();
        return true;
    }

    public async Task<string> UploadAsync(Stream stream, string fileName, string targetPath)
    {
        var folderId = await ResolveFolderIdByPath(targetPath);

        var fileMetadata = new Google.Apis.Drive.v3.Data.File
        {
            Name = fileName,
            Parents = new List<string> { folderId }
        };

        var request = _driveService.Files.Create(fileMetadata, stream, GetMimeType(fileName));
        request.Fields = "id, webViewLink";
        var uploadResult = await request.UploadAsync();

        return request.ResponseBody?.Id ?? "";
    }

    public async Task<Stream> DownloadAsync(string fileId)
    {
        var stream = new MemoryStream();
        await _driveService.Files.Get(fileId).DownloadAsync(stream);
        stream.Position = 0;
        return stream;
    }

    // Helper methods...
    private async Task<string> ResolveFolderIdByPath(string path) { /* ... */ }
    private List<BreadcrumbDto> BuildBreadcrumbs(string path) { /* ... */ }
    private string GetOrderBy(string sortBy, bool isDesc) { /* ... */ }
    private string GetMimeType(string fileName) { /* ... */ }
    private async Task<(int files, int folders)> CountItemsRecursive(string folderId) { /* ... */ }
}

6. Server Local Storage (Disk Management)

📁 Services/Storage/LocalStorageProvider.cs
using System.IO;

public class LocalStorageProvider : IStorageProvider
{
    private readonly string _rootPath;
    private readonly long _maxSizeBytes;

    public string ProviderId => "server";
    public string Label => "Server Storage";

    public LocalStorageProvider(IConfiguration config)
    {
        _rootPath = config["Storage:Local:RootPath"] ?? "/data/xmtruyen";
        var maxGB = config.GetValue<int>("Storage:Local:MaxSizeGB", 50);
        _maxSizeBytes = (long)maxGB * 1024 * 1024 * 1024;
    }

    public Task<StorageOverviewDto> GetOverviewAsync()
    {
        var driveInfo = new DriveInfo(Path.GetPathRoot(_rootPath)!);

        // Tính dung lượng thực tế của thư mục root
        var dirInfo = new DirectoryInfo(_rootPath);
        var usedSpace = GetDirectorySize(dirInfo);
        var (fileCount, folderCount) = CountItems(dirInfo);

        return Task.FromResult(new StorageOverviewDto
        {
            Provider = ProviderId,
            Label = Label,
            TotalSpace = Math.Min(_maxSizeBytes, driveInfo.TotalSize),
            UsedSpace = usedSpace,
            FreeSpace = Math.Min(_maxSizeBytes - usedSpace, driveInfo.AvailableFreeSpace),
            TotalFiles = fileCount,
            TotalFolders = folderCount
        });
    }

    public Task<StorageListResult> ListFilesAsync(string path, string? search, string sortBy, bool isDesc)
    {
        var fullPath = Path.Combine(_rootPath, path.TrimStart('/'));
        var dirInfo = new DirectoryInfo(fullPath);

        if (!dirInfo.Exists)
            return Task.FromResult(new StorageListResult { Items = new(), CurrentPath = path });

        var items = new List<StorageFileItemDto>();

        // Thêm folders
        foreach (var dir in dirInfo.GetDirectories())
        {
            if (!string.IsNullOrEmpty(search) && !dir.Name.Contains(search, StringComparison.OrdinalIgnoreCase))
                continue;

            items.Add(new StorageFileItemDto
            {
                Id = GetRelativePath(dir.FullName),
                Name = dir.Name,
                Path = GetRelativePath(dir.FullName),
                Type = "folder",
                Size = 0,
                CreatedAt = dir.CreationTimeUtc,
                ModifiedAt = dir.LastWriteTimeUtc,
                ChildCount = dir.GetFileSystemInfos().Length
            });
        }

        // Thêm files
        foreach (var file in dirInfo.GetFiles())
        {
            if (!string.IsNullOrEmpty(search) && !file.Name.Contains(search, StringComparison.OrdinalIgnoreCase))
                continue;

            items.Add(new StorageFileItemDto
            {
                Id = GetRelativePath(file.FullName),
                Name = file.Name,
                Path = GetRelativePath(file.FullName),
                Type = "file",
                Size = file.Length,
                MimeType = GetMimeType(file.Extension),
                CreatedAt = file.CreationTimeUtc,
                ModifiedAt = file.LastWriteTimeUtc
            });
        }

        // Sort
        items = sortBy switch
        {
            "size" => isDesc ? items.OrderByDescending(i => i.Size).ToList() : items.OrderBy(i => i.Size).ToList(),
            "createdAt" => isDesc ? items.OrderByDescending(i => i.CreatedAt).ToList() : items.OrderBy(i => i.CreatedAt).ToList(),
            _ => isDesc ? items.OrderByDescending(i => i.Name).ToList() : items.OrderBy(i => i.Name).ToList()
        };

        return Task.FromResult(new StorageListResult
        {
            Items = items,
            TotalCount = items.Count,
            CurrentPath = path,
            Breadcrumbs = BuildBreadcrumbs(path)
        });
    }

    public Task<bool> DeleteAsync(string itemId)
    {
        var fullPath = Path.Combine(_rootPath, itemId.TrimStart('/'));

        if (Directory.Exists(fullPath))
        {
            Directory.Delete(fullPath, recursive: true);
            return Task.FromResult(true);
        }
        else if (File.Exists(fullPath))
        {
            File.Delete(fullPath);
            return Task.FromResult(true);
        }

        return Task.FromResult(false);
    }

    public async Task<string> UploadAsync(Stream stream, string fileName, string targetPath)
    {
        var fullDir = Path.Combine(_rootPath, targetPath.TrimStart('/'));
        Directory.CreateDirectory(fullDir);

        var filePath = Path.Combine(fullDir, fileName);
        using var fileStream = new FileStream(filePath, FileMode.Create);
        await stream.CopyToAsync(fileStream);

        return GetRelativePath(filePath);
    }

    public Task<Stream> DownloadAsync(string itemId)
    {
        var fullPath = Path.Combine(_rootPath, itemId.TrimStart('/'));
        Stream stream = new FileStream(fullPath, FileMode.Open, FileAccess.Read);
        return Task.FromResult(stream);
    }

    // Helpers
    private long GetDirectorySize(DirectoryInfo dir)
    {
        return dir.EnumerateFiles("*", SearchOption.AllDirectories).Sum(f => f.Length);
    }

    private (int files, int folders) CountItems(DirectoryInfo dir)
    {
        var files = dir.EnumerateFiles("*", SearchOption.AllDirectories).Count();
        var folders = dir.EnumerateDirectories("*", SearchOption.AllDirectories).Count();
        return (files, folders);
    }

    private string GetRelativePath(string fullPath)
    {
        return "/" + Path.GetRelativePath(_rootPath, fullPath).Replace("\\", "/");
    }

    private List<BreadcrumbDto> BuildBreadcrumbs(string path)
    {
        var parts = path.Split('/', StringSplitOptions.RemoveEmptyEntries);
        var breadcrumbs = new List<BreadcrumbDto>();
        for (int i = 0; i < parts.Length; i++)
        {
            breadcrumbs.Add(new BreadcrumbDto
            {
                Name = parts[i],
                Path = "/" + string.Join("/", parts.Take(i + 1))
            });
        }
        return breadcrumbs;
    }

    private string GetMimeType(string ext) => ext.ToLower() switch
    {
        ".jpg" or ".jpeg" => "image/jpeg",
        ".png" => "image/png",
        ".gif" => "image/gif",
        ".webp" => "image/webp",
        ".pdf" => "application/pdf",
        ".epub" => "application/epub+zip",
        ".mp3" => "audio/mpeg",
        ".mp4" => "video/mp4",
        ".json" => "application/json",
        ".zip" => "application/zip",
        _ => "application/octet-stream"
    };
}

9.2 Controller Code

📁 Controllers/StorageController.cs
[ApiController]
[Route("api/[controller]")]
[Authorize(Roles = "Admin")]
public class StorageController : ControllerBase
{
    private readonly IEnumerable<IStorageProvider> _providers;
    private readonly AppDbContext _db;

    public StorageController(IEnumerable<IStorageProvider> providers, AppDbContext db)
    {
        _providers = providers;
        _db = db;
    }

    [HttpGet("overview/all")]
    public async Task<IActionResult> GetAllOverview()
    {
        var results = new List<StorageOverviewDto>();
        foreach (var provider in _providers)
        {
            results.Add(await provider.GetOverviewAsync());
        }
        return Ok(new { success = true, data = results });
    }

    [HttpGet("files")]
    public async Task<IActionResult> ListFiles(
        [FromQuery] string provider,
        [FromQuery] string path = "/",
        [FromQuery] string? search = null,
        [FromQuery] string sortBy = "name",
        [FromQuery] bool isDescending = false)
    {
        var storageProvider = GetProvider(provider);
        var result = await storageProvider.ListFilesAsync(path, search, sortBy, isDescending);

        return Ok(new
        {
            success = true,
            data = result.Items,
            totalCount = result.TotalCount,
            currentPath = result.CurrentPath,
            breadcrumbs = result.Breadcrumbs
        });
    }

    [HttpPost("move")]
    public async Task<IActionResult> MoveFile([FromBody] MoveFileRequest request)
    {
        var source = GetProvider(request.SourceProvider);
        var target = GetProvider(request.TargetProvider);

        // Tạo log
        var log = new StorageTransferLog
        {
            SourceProvider = request.SourceProvider,
            TargetProvider = request.TargetProvider,
            FileName = request.ItemId, // sẽ update sau khi lấy metadata
            Status = "transferring"
        };
        _db.StorageTransferLogs.Add(log);
        await _db.SaveChangesAsync();

        try
        {
            // Download từ source
            using var stream = await source.DownloadAsync(request.ItemId);

            // Upload lên target
            var targetPath = request.TargetPath ?? "/";
            var fileName = Path.GetFileName(request.ItemId);
            await target.UploadAsync(stream, fileName, targetPath);

            // Xóa file gốc (move, không phải copy)
            await source.DeleteAsync(request.ItemId);

            // Update log
            log.Status = "completed";
            log.CompletedAt = DateTime.UtcNow;
            await _db.SaveChangesAsync();

            return Ok(new { success = true, message = $"Đã di chuyển '{fileName}' thành công." });
        }
        catch (Exception ex)
        {
            log.Status = "failed";
            log.ErrorMessage = ex.Message;
            await _db.SaveChangesAsync();

            return StatusCode(500, new { success = false, message = ex.Message });
        }
    }

    [HttpDelete("files/{itemId}")]
    public async Task<IActionResult> DeleteFile(string itemId, [FromQuery] string provider)
    {
        var storageProvider = GetProvider(provider);
        var result = await storageProvider.DeleteAsync(Uri.UnescapeDataString(itemId));
        return Ok(new { success = result, message = result ? "Đã xóa." : "Không tìm thấy file." });
    }

    [HttpPost("upload")]
    public async Task<IActionResult> Upload([FromForm] IFormFile file, [FromForm] string provider, [FromForm] string? path)
    {
        var storageProvider = GetProvider(provider);
        using var stream = file.OpenReadStream();
        var resultPath = await storageProvider.UploadAsync(stream, file.FileName, path ?? "/");
        return Ok(new { success = true, url = resultPath, message = "Upload thành công." });
    }

    private IStorageProvider GetProvider(string providerId) =>
        _providers.First(p => p.ProviderId == providerId);
}

public record MoveFileRequest(string SourceProvider, string TargetProvider, string ItemId, string? TargetPath);

10. Monitoring & Cảnh Báo Dung Lượng

10.1 Background Job — Kiểm tra dung lượng định kỳ

📁 Jobs/StorageMonitorJob.cs
/// <summary>
/// Chạy mỗi 1 giờ — kiểm tra dung lượng, tạo alert nếu gần đầy
/// </summary>
public class StorageMonitorJob : BackgroundService
{
    private readonly IEnumerable<IStorageProvider> _providers;
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly ILogger<StorageMonitorJob> _logger;

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            await CheckStorageSpace();
            await Task.Delay(TimeSpan.FromHours(1), stoppingToken);
        }
    }

    private async Task CheckStorageSpace()
    {
        using var scope = _scopeFactory.CreateScope();
        var db = scope.ServiceProvider.GetRequiredService<AppDbContext>();

        foreach (var provider in _providers)
        {
            try
            {
                var overview = await provider.GetOverviewAsync();
                var usedPercent = (double)overview.UsedSpace / overview.TotalSpace * 100;

                string? alertType = usedPercent switch
                {
                    >= 95 => "space_full",
                    >= 90 => "space_critical_90",
                    >= 80 => "space_warning_80",
                    _ => null
                };

                if (alertType != null)
                {
                    // Kiểm tra alert đã tồn tại chưa (tránh spam)
                    var existingAlert = await db.StorageAlerts
                        .Where(a => a.ProviderId == provider.ProviderId
                                 && a.AlertType == alertType
                                 && !a.IsAcknowledged
                                 && a.CreatedAt > DateTime.UtcNow.AddHours(-24))
                        .FirstOrDefaultAsync();

                    if (existingAlert == null)
                    {
                        db.StorageAlerts.Add(new StorageAlert
                        {
                            ProviderId = provider.ProviderId,
                            AlertType = alertType,
                            Message = $"{provider.Label}: Dung lượng đã sử dụng {usedPercent:F1}% " +
                                     $"({FormatBytes(overview.UsedSpace)} / {FormatBytes(overview.TotalSpace)})",
                            UsedPercent = (decimal)usedPercent
                        });
                        await db.SaveChangesAsync();

                        _logger.LogWarning("Storage alert: {Provider} at {Percent}%",
                            provider.ProviderId, usedPercent);
                    }
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error checking storage for {Provider}", provider.ProviderId);
            }
        }
    }
}

11.1 Path Traversal Protection

private string SanitizePath(string path)
{
    // Chặn path traversal
    if (path.Contains("..") || path.Contains("~"))
        throw new ArgumentException("Invalid path: directory traversal not allowed");

    // Normalize
    path = path.Replace("\\", "/").TrimStart('/');

    // Đảm bảo nằm trong root
    var fullPath = Path.GetFullPath(Path.Combine(_rootPath, path));
    if (!fullPath.StartsWith(_rootPath))
        throw new UnauthorizedAccessException("Access denied: path outside storage root");

    return fullPath;
}
