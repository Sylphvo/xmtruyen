2.1 Danh Sách Bảng Mới

Bảng Khóa Chính Quan Hệ Mô Tả
Courses Id (Guid PK) InstructorId → Users Thông tin khóa học: tên, mô tả, ảnh bìa, giá, trạng thái (Draft/Published/Archived)
CourseSections Id (Guid PK) CourseId → Courses Phân mục trong khóa học (ví dụ: "Chương 1: Giới thiệu")

3.2 Upload Controller

📁 xomtruyen.API/Controllers/CourseVideoController.cs
[Route("api/courses/videos")]
[ApiController]
[Authorize(Roles = "Admin,Instructor")]
public class CourseVideoController : ControllerBase
{
    private readonly IVideoService _videoService;
    private readonly IBackgroundTaskQueue _taskQueue;

    public CourseVideoController(IVideoService videoService, IBackgroundTaskQueue taskQueue)
    {
        _videoService = videoService;
        _taskQueue = taskQueue;
    }

    /// <summary>
    /// Upload video bài học (hỗ trợ file lớn qua chunked upload)
    /// </summary>
    [HttpPost("upload")]
    [RequestSizeLimit(2_147_483_648)] // 2GB max
    [RequestFormLimits(MultipartBodyLengthLimit = 2_147_483_648)]
    public async Task<IActionResult> UploadLessonVideo(
        IFormFile file,
        [FromForm] Guid lessonId)
    {
        if (file == null || file.Length == 0)
            return BadRequest("File video không hợp lệ");

        // Validate video format
        var allowedExtensions = new[] { ".mp4", ".mkv", ".webm", ".mov", ".avi" };
        var ext = Path.GetExtension(file.FileName).ToLower();
        if (!allowedExtensions.Contains(ext))
            return BadRequest($"Định dạng {ext} không được hỗ trợ");

        // 1. Lưu file gốc
        var savedPath = await _videoService.SaveOriginalAsync(file, lessonId);

        // 2. Đẩy vào hàng đợi xử lý nền (transcode → multiple resolutions)
        await _taskQueue.QueueBackgroundWorkItemAsync(new VideoProcessingTask
        {
            TaskId = Guid.NewGuid().ToString(),
            LessonId = lessonId,
            SourcePath = savedPath,
            OriginalFileName = file.FileName,
            FileSize = file.Length
        });

        return Accepted(new
        {
            message = "Video đã được nhận, đang xử lý nền",
            lessonId = lessonId,
            fileName = file.FileName,
            fileSize = file.Length
        });
    }

    /// <summary>
    /// Stream video cho học viên (có kiểm tra quyền truy cập)
    /// </summary>
    [HttpGet("stream/{lessonId}")]
    [Authorize]
    public async Task<IActionResult> StreamVideo(Guid lessonId, [FromQuery] string quality = "720p")
    {
        var userId = GetCurrentUserId();
        var hasAccess = await _videoService.CheckAccessAsync(userId, lessonId);
        if (!hasAccess)
            return Forbid("Bạn chưa đăng ký khóa học này");

        var videoPath = await _videoService.GetVideoPathAsync(lessonId, quality);
        if (string.IsNullOrEmpty(videoPath) || !System.IO.File.Exists(videoPath))
            return NotFound("Video không tồn tại");

        var stream = new FileStream(videoPath, FileMode.Open, FileAccess.Read);
        return File(stream, "video/mp4", enableRangeProcessing: true);
    }
}

3.3 Video Processing Worker (Transcode)

📁 xomtruyen.API/Services/Workers/VideoProcessingWorker.cs
public class VideoProcessingWorker : BackgroundService
{
    private readonly IBackgroundTaskQueue _taskQueue;
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly ILogger<VideoProcessingWorker> _logger;

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            var task = await _taskQueue.DequeueAsync(stoppingToken);
            if (task is VideoProcessingTask videoTask)
            {
                await ProcessVideoAsync(videoTask);
            }
        }
    }

    private async Task ProcessVideoAsync(VideoProcessingTask task)
    {
        _logger.LogInformation("Bắt đầu xử lý video: {FileName}", task.OriginalFileName);

        using var scope = _scopeFactory.CreateScope();
        var db = scope.ServiceProvider.GetRequiredService<ApplicationDbContext>();

        try
        {
            // 1. Trích xuất metadata (duration, resolution, codec)
            var metadata = await ExtractMetadataAsync(task.SourcePath);

            // 2. Transcode sang nhiều chất lượng (480p, 720p, 1080p)
            var outputs = new Dictionary<string, string>();
            foreach (var quality in new[] { "480p", "720p", "1080p" })
            {
                var outputPath = await TranscodeAsync(task.SourcePath, quality);
                outputs[quality] = outputPath;
            }

            // 3. Tạo thumbnail từ frame giữa video
            var thumbnailPath = await GenerateThumbnailAsync(task.SourcePath);

            // 4. Lưu thông tin vào DB
            var lessonVideo = new LessonVideo
            {
                LessonId = Guid.Parse(task.LessonId.ToString()),
                OriginalUrl = task.SourcePath,
                Url480p = outputs.GetValueOrDefault("480p"),
                Url720p = outputs.GetValueOrDefault("720p"),
                Url1080p = outputs.GetValueOrDefault("1080p"),
                ThumbnailUrl = thumbnailPath,
                DurationSeconds = metadata.DurationSeconds,
                FileSizeBytes = task.FileSize,
                Resolution = metadata.Resolution,
                Codec = metadata.Codec,
                Status = VideoStatus.Ready
            };

            db.LessonVideos.Add(lessonVideo);

            // Cập nhật duration cho Lesson
            var lesson = await db.Lessons.FindAsync(task.LessonId);
            if (lesson != null)
                lesson.DurationSeconds = metadata.DurationSeconds;

            await db.SaveChangesAsync();
            _logger.LogInformation("✅ Video xử lý xong: {FileName}", task.OriginalFileName);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "❌ Lỗi xử lý video: {FileName}", task.OriginalFileName);
        }
    }

    /// <summary> Sử dụng FFmpeg để transcode video </summary>
    private async Task<string> TranscodeAsync(string inputPath, string quality)
    {
        var resolutions = new Dictionary<string, string>
        {
            ["480p"] = "854:480",
            ["720p"] = "1280:720",
            ["1080p"] = "1920:1080"
        };

        var outputDir = Path.GetDirectoryName(inputPath)!;
        var outputFile = Path.Combine(outputDir, $"{Path.GetFileNameWithoutExtension(inputPath)}_{quality}.mp4");

        var args = $"-i \"{inputPath}\" -vf scale={resolutions[quality]} " +
                   $"-c:v libx264 -preset medium -crf 23 -c:a aac -b:a 128k " +
                   $"-movflags +faststart \"{outputFile}\"";

        await RunFFmpegAsync(args);
        return outputFile;
    }
}

4. Video Player & Theo Dõi Tiến Độ

4.1 Kiến Trúc Player

  • Thư viện: Video.js hoặc Plyr (hỗ trợ HLS adaptive streaming)
  • Tính năng: Chuyển chất lượng (480p/720p/1080p), tốc độ phát (0.5x → 2x), Picture-in-Picture, Fullscreen
  • Theo dõi: Mỗi 10 giây gửi heartbeat API cập nhật vị trí xem → LessonProgress
  • Hoàn thành: Khi xem ≥90% video → đánh dấu IsCompleted = true
  • Bảo mật: Video stream qua API có token, không expose URL trực tiếp

4.2 React Video Player Component

📁 xom-truyen/src/components/Course/VideoPlayer.tsx
import React, { useRef, useEffect, useCallback, useState } from 'react';
import axios from 'axios';

interface VideoPlayerProps {
    lessonId: string;
    videoUrl: string;
    initialPosition?: number;  // Giây bắt đầu (resume)
    onComplete?: () => void;
}

export default function VideoPlayer({ lessonId, videoUrl, initialPosition = 0, onComplete }: VideoPlayerProps) {
    const videoRef = useRef<HTMLVideoElement>(null);
    const [currentTime, setCurrentTime] = useState(0);
    const [duration, setDuration] = useState(0);
    const [quality, setQuality] = useState('720p');
    const [playbackRate, setPlaybackRate] = useState(1);
    const progressInterval = useRef<number>();

    // Gửi tiến độ lên server mỗi 10 giây
    const saveProgress = useCallback(async () => {
        if (!videoRef.current || !duration) return;

        const percent = (videoRef.current.currentTime / duration) * 100;
        const position = Math.floor(videoRef.current.currentTime);

        try {
            await axios.post('/api/courses/progress/update', {
                lessonId,
                watchedPercent: Math.min(percent, 100),
                lastPositionSeconds: position,
                isCompleted: percent >= 90
            });

            // Nếu xem >= 90% → gọi callback hoàn thành
            if (percent >= 90 && onComplete) {
                onComplete();
            }
        } catch (err) {
            console.warn('Không thể lưu tiến độ:', err);
        }
    }, [lessonId, duration, onComplete]);

    useEffect(() => {
        // Bắt đầu interval gửi progress
        progressInterval.current = window.setInterval(saveProgress, 10000);
        return () => {
            clearInterval(progressInterval.current);
            saveProgress(); // Lưu lần cuối khi unmount
        };
    }, [saveProgress]);

    useEffect(() => {
        // Resume từ vị trí cuối
        if (videoRef.current && initialPosition > 0) {
            videoRef.current.currentTime = initialPosition;
        }
    }, [initialPosition]);

    const handleQualityChange = (newQuality: string) => {
        const currentPos = videoRef.current?.currentTime || 0;
        setQuality(newQuality);
        // URL sẽ thay đổi theo quality, video tự load lại
        setTimeout(() => {
            if (videoRef.current) videoRef.current.currentTime = currentPos;
        }, 500);
    };

    return (
        <div className="video-player-container">
            <video
                ref={videoRef}
                src={`${videoUrl}?quality=${quality}`}
                controls
                onTimeUpdate={() => setCurrentTime(videoRef.current?.currentTime || 0)}
                onLoadedMetadata={() => setDuration(videoRef.current?.duration || 0)}
                style={{ width: '100%', borderRadius: '8px', background: '#000' }}
            />

            {/* Controls Bar */}
            <div className="player-controls">
                {/* Quality Selector */}
                <select value={quality} onChange={e => handleQualityChange(e.target.value)}>
                    <option value="480p">480p</option>
                    <option value="720p">720p (HD)</option>
                    <option value="1080p">1080p (Full HD)</option>
                </select>

                {/* Playback Speed */}
                <select value={playbackRate} onChange={e => {
                    const rate = parseFloat(e.target.value);
                    setPlaybackRate(rate);
                    if (videoRef.current) videoRef.current.playbackRate = rate;
                }}>
                    <option value="0.5">0.5x</option>
                    <option value="0.75">0.75x</option>
                    <option value="1">1x (Bình thường)</option>
                    <option value="1.25">1.25x</option>
                    <option value="1.5">1.5x</option>
                    <option value="2">2x</option>
                </select>

                {/* Progress Info */}
                <span className="progress-info">
                    {Math.floor(currentTime / 60)}:{String(Math.floor(currentTime % 60)).padStart(2, '0')}
                    {' / '}
                    {Math.floor(duration / 60)}:{String(Math.floor(duration % 60)).padStart(2, '0')}
                    {' • '}
                    {duration ? Math.floor((currentTime / duration) * 100) : 0}% hoàn thành
                </span>
            </div>
        </div>
    );
}

📁 xomtruyen.API/Controllers/CourseProgressController.cs
[Route("api/courses/progress")]
[ApiController]
[Authorize]
public class CourseProgressController : ControllerBase
{
    private readonly ApplicationDbContext _db;

    [HttpPost("update")]
    public async Task<IActionResult> UpdateProgress([FromBody] UpdateProgressRequest request)
    {
        var userId = GetCurrentUserId();

        var progress = await _db.LessonProgresses
            .FirstOrDefaultAsync(p => p.UserId == userId && p.LessonId == request.LessonId);

        if (progress == null)
        {
            progress = new LessonProgress
            {
                UserId = userId,
                LessonId = request.LessonId
            };
            _db.LessonProgresses.Add(progress);
        }

        progress.WatchedPercent = Math.Max(progress.WatchedPercent, request.WatchedPercent);
        progress.LastPositionSeconds = request.LastPositionSeconds;
        progress.LastAccessedAt = DateTime.UtcNow;

        if (request.IsCompleted && !progress.IsCompleted)
        {
            progress.IsCompleted = true;
            progress.CompletedAt = DateTime.UtcNow;
        }

        await _db.SaveChangesAsync();
        return Ok(new { success = true });
    }

    [HttpGet("course/{courseId}")]
    public async Task<IActionResult> GetCourseProgress(Guid courseId)
    {
        var userId = GetCurrentUserId();

        var lessons = await _db.Lessons
            .Where(l => l.Section.CourseId == courseId)
            .Select(l => l.Id)
            .ToListAsync();

        var progresses = await _db.LessonProgresses
            .Where(p => p.UserId == userId && lessons.Contains(p.LessonId))
            .ToListAsync();

        var totalLessons = lessons.Count;
        var completedLessons = progresses.Count(p => p.IsCompleted);
        var overallPercent = totalLessons > 0 ? (double)completedLessons / totalLessons * 100 : 0;

        return Ok(new
        {
            courseId,
            totalLessons,
            completedLessons,
            overallPercent = Math.Round(overallPercent, 1),
            lessons = progresses.Select(p => new
            {
                p.LessonId,
                p.WatchedPercent,
                p.LastPositionSeconds,
                p.IsCompleted,
                p.LastAccessedAt
            })
        });
    }
}

📁 xomtruyen.API/Controllers/CoursePaymentController.cs
[Route("api/courses/payment")]
[ApiController]
[Authorize]
public class CoursePaymentController : ControllerBase
{
    private readonly IPaymentService _paymentService;
    private readonly ApplicationDbContext _db;

    /// <summary>
    /// Tạo đơn thanh toán và redirect đến cổng thanh toán
    /// </summary>
    [HttpPost("create")]
    public async Task<IActionResult> CreatePayment([FromBody] CreatePaymentRequest request)
    {
        var userId = GetCurrentUserId();

        // Kiểm tra đã mua chưa
        var existing = await _db.CourseEnrollments
            .AnyAsync(e => e.UserId == userId && e.CourseId == request.CourseId);
        if (existing)
            return BadRequest("Bạn đã đăng ký khóa học này rồi");

        var course = await _db.Courses.FindAsync(request.CourseId);
        if (course == null) return NotFound("Khóa học không tồn tại");

        // Tính giá (áp dụng giảm giá nếu có)
        var amount = course.DiscountPrice ?? course.Price;

        // Nếu miễn phí → tự động enroll
        if (amount == 0)
        {
            await EnrollUser(userId, request.CourseId);
            return Ok(new { success = true, message = "Đăng ký thành công (miễn phí)", redirectUrl = "" });
        }

        // Tạo payment record
        var payment = new CoursePayment
        {
            UserId = userId,
            CourseId = request.CourseId,
            Amount = amount,
            Method = request.Method,
            Status = PaymentStatus.Pending
        };
        _db.CoursePayments.Add(payment);
        await _db.SaveChangesAsync();

        // Tạo URL thanh toán từ cổng
        var paymentUrl = request.Method switch
        {
            PaymentMethod.VNPay => await _paymentService.CreateVNPayUrlAsync(payment),
            PaymentMethod.MoMo => await _paymentService.CreateMoMoUrlAsync(payment),
            _ => throw new Exception("Phương thức thanh toán không hỗ trợ")
        };

        return Ok(new { success = true, paymentId = payment.Id, redirectUrl = paymentUrl });
    }

    /// <summary>
    /// Callback từ VNPay (IPN - Instant Payment Notification)
    /// </summary>
    [HttpGet("vnpay-callback")]
    [AllowAnonymous]
    public async Task<IActionResult> VNPayCallback([FromQuery] Dictionary<string, string> queryParams)
    {
        var isValid = _paymentService.ValidateVNPaySignature(queryParams);
        if (!isValid)
            return BadRequest("Chữ ký không hợp lệ");

        var paymentId = Guid.Parse(queryParams["vnp_TxnRef"]);
        var payment = await _db.CoursePayments.FindAsync(paymentId);
        if (payment == null) return NotFound();

        var responseCode = queryParams["vnp_ResponseCode"];
        if (responseCode == "00") // Thành công
        {
            payment.Status = PaymentStatus.Completed;
            payment.TransactionCode = queryParams["vnp_TransactionNo"];
            payment.PaidAt = DateTime.UtcNow;
            payment.GatewayResponse = System.Text.Json.JsonSerializer.Serialize(queryParams);

            // Tự động tạo enrollment
            await EnrollUser(payment.UserId, payment.CourseId);
        }
        else
        {
            payment.Status = PaymentStatus.Failed;
            payment.GatewayResponse = System.Text.Json.JsonSerializer.Serialize(queryParams);
        }

        await _db.SaveChangesAsync();

        // Redirect về trang kết quả trên client
        var clientUrl = responseCode == "00"
            ? $"http://localhost:5173/courses/{payment.CourseId}/success"
            : $"http://localhost:5173/courses/{payment.CourseId}/failed";

        return Redirect(clientUrl);
    }

    private async Task EnrollUser(Guid userId, Guid courseId)
    {
        var enrollment = new CourseEnrollment
        {
            UserId = userId,
            CourseId = courseId,
            EnrolledAt = DateTime.UtcNow,
            Status = EnrollmentStatus.Active
        };
        _db.CourseEnrollments.Add(enrollment);

        // Tăng enrollment count
        var course = await _db.Courses.FindAsync(courseId);
        if (course != null) course.EnrollmentCount++;

        await _db.SaveChangesAsync();
    }
}

5.3 VNPay Service

📁 xomtruyen.API/Services/Implementations/VNPayService.cs
public class VNPayService : IPaymentService
{
    private readonly IConfiguration _config;

    public async Task<string> CreateVNPayUrlAsync(CoursePayment payment)
    {
        var vnp_Url = _config["VNPay:Url"];           // https://sandbox.vnpayment.vn/paymentv2/vpcpay.html
        var vnp_TmnCode = _config["VNPay:TmnCode"];   // Mã website tại VNPay
        var vnp_HashSecret = _config["VNPay:HashSecret"];
        var vnp_ReturnUrl = _config["VNPay:ReturnUrl"]; // /api/courses/payment/vnpay-callback

        var vnpay = new SortedDictionary<string, string>
        {
            ["vnp_Version"] = "2.1.0",
            ["vnp_Command"] = "pay",
            ["vnp_TmnCode"] = vnp_TmnCode,
            ["vnp_Amount"] = ((long)(payment.Amount * 100)).ToString(), // VNPay nhận đơn vị: đồng * 100
            ["vnp_CurrCode"] = "VND",
            ["vnp_TxnRef"] = payment.Id.ToString(),
            ["vnp_OrderInfo"] = $"Thanh toan khoa hoc - {payment.CourseId}",
            ["vnp_OrderType"] = "education",
            ["vnp_Locale"] = "vn",
            ["vnp_ReturnUrl"] = vnp_ReturnUrl,
            ["vnp_IpAddr"] = "127.0.0.1",
            ["vnp_CreateDate"] = DateTime.Now.ToString("yyyyMMddHHmmss")
        };

        // Tạo chuỗi ký (HMAC SHA512)
        var signData = string.Join("&", vnpay.Select(kv => $"{kv.Key}={kv.Value}"));
        var vnp_SecureHash = HmacSHA512(vnp_HashSecret, signData);
        var paymentUrl = $"{vnp_Url}?{signData}&vnp_SecureHash={vnp_SecureHash}";

        return paymentUrl;
    }

    private string HmacSHA512(string key, string data)
    {
        using var hmac = new System.Security.Cryptography.HMACSHA512(Encoding.UTF8.GetBytes(key));
        var hash = hmac.ComputeHash(Encoding.UTF8.GetBytes(data));
        return BitConverter.ToString(hash).Replace("-", "").ToLower();
    }
}
