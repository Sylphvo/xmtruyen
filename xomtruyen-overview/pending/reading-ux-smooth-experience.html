2.1 Lưu trữ Settings (LocalStorage)

📁 xom-truyen/src/hooks/useReadingSettings.ts
import { useState, useEffect, useCallback } from 'react';

interface ReadingSettings {
    fontSize: number;         // 14-24px
    fontFamily: string;       // 'Roboto' | 'Literata' | 'Merriweather'
    lineHeight: number;       // 1.4 - 2.2
    theme: 'light' | 'dark' | 'sepia';
    contentWidth: number;     // 600 | 720 | 900
    autoScrollSpeed: number;  // 0 (off), 1-8 px/s
}

const DEFAULT_SETTINGS: ReadingSettings = {
    fontSize: 18,
    fontFamily: 'Roboto',
    lineHeight: 1.8,
    theme: 'light',
    contentWidth: 720,
    autoScrollSpeed: 0,
};

export function useReadingSettings() {
    const [settings, setSettings] = useState<ReadingSettings>(() => {
        const saved = localStorage.getItem('xomtruyen_reading_settings');
        return saved ? { ...DEFAULT_SETTINGS, ...JSON.parse(saved) } : DEFAULT_SETTINGS;
    });

    // Auto-save khi thay đổi
    useEffect(() => {
        localStorage.setItem('xomtruyen_reading_settings', JSON.stringify(settings));
    }, [settings]);

    const updateSetting = useCallback(<K extends keyof ReadingSettings>(key: K, value: ReadingSettings[K]) => {
        setSettings(prev => ({ ...prev, [key]: value }));
    }, []);

    const resetSettings = useCallback(() => setSettings(DEFAULT_SETTINGS), []);

    // CSS variables áp dụng lên content
    const contentStyle: React.CSSProperties = {
        fontSize: `${settings.fontSize}px`,
        fontFamily: settings.fontFamily,
        lineHeight: settings.lineHeight,
        maxWidth: `${settings.contentWidth}px`,
        backgroundColor: settings.theme === 'dark' ? '#1a1a2e' 
                        : settings.theme === 'sepia' ? '#f5e6d3' : '#ffffff',
        color: settings.theme === 'dark' ? '#e0e0e0' 
             : settings.theme === 'sepia' ? '#5b4636' : '#374151',
        transition: 'background-color 400ms ease, color 300ms ease, font-size 200ms ease',
    };

    return { settings, updateSetting, resetSettings, contentStyle };
}

3. Thanh Tiến Độ & Resume Đọc

📁 xom-truyen/src/components/Reading/ReadingProgressBar.tsx
import React, { useState, useEffect, useRef } from 'react';

interface Props {
    chapterId: string;
    onProgressChange?: (percent: number) => void;
}

/**
 * Thanh tiến độ đọc — sticky top 3px.
 * Tự động tính % dựa trên scroll position.
 * Auto-save mỗi 10 giây + resume khi quay lại.
 */
export const ReadingProgressBar: React.FC<Props> = ({ chapterId, onProgressChange }) => {
    const [progress, setProgress] = useState(0);
    const saveTimerRef = useRef<number>();

    // Tính % scroll
    useEffect(() => {
        const handleScroll = () => {
            const scrollTop = window.scrollY;
            const docHeight = document.documentElement.scrollHeight - window.innerHeight;
            const percent = docHeight > 0 ? Math.min((scrollTop / docHeight) * 100, 100) : 0;
            setProgress(percent);
            onProgressChange?.(percent);
        };

        window.addEventListener('scroll', handleScroll, { passive: true });
        return () => window.removeEventListener('scroll', handleScroll);
    }, [onProgressChange]);

    // Auto-save mỗi 10 giây
    useEffect(() => {
        saveTimerRef.current = window.setInterval(() => {
            if (progress > 0) {
                localStorage.setItem(`reading_pos_${chapterId}`, String(progress));
                // Gửi lên server (background, không block)
                fetch('/api/reading/progress', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ chapterId, scrollPercent: progress }),
                    keepalive: true
                }).catch(() => {});
            }
        }, 10000);

        return () => clearInterval(saveTimerRef.current);
    }, [chapterId, progress]);

    // Resume position khi mount
    useEffect(() => {
        const saved = localStorage.getItem(`reading_pos_${chapterId}`);
        if (saved) {
            const percent = parseFloat(saved);
            if (percent > 5) { // Chỉ resume nếu > 5%
                requestAnimationFrame(() => {
                    const docHeight = document.documentElement.scrollHeight - window.innerHeight;
                    window.scrollTo({ top: (percent / 100) * docHeight, behavior: 'smooth' });
                });
            }
        }
    }, [chapterId]);

    return (
        <div style={{
            position: 'sticky',
            top: 0,
            left: 0,
            width: '100%',
            height: '3px',
            backgroundColor: 'rgba(0,0,0,0.05)',
            zIndex: 1000,
            overflow: 'hidden'
        }}>
            <div style={{
                height: '100%',
                width: `${progress}%`,
                backgroundColor: '#2196f3',
                transition: 'width 100ms linear',
                borderRadius: '0 2px 2px 0'
            }} />
        </div>
    );
};

4.1 Animations cho History Items

📁 xom-truyen/src/components/History/HistoryList.tsx
import React, { useEffect, useState } from 'react';
import { useNavigate } from 'react-router-dom';

interface HistoryItem {
    id: string;
    publication: { id: string; title: string; coverImageUrl: string; formatType: number };
    chapterTitle: string;
    chapterNumber: number;
    scrollPercent: number;
    updatedAt: string;
}

export const HistoryList: React.FC<{ items: HistoryItem[] }> = ({ items }) => {
    const navigate = useNavigate();
    const [visibleItems, setVisibleItems] = useState<Set<string>>(new Set());
    const [removingId, setRemovingId] = useState<string | null>(null);

    // Stagger animation: mỗi item xuất hiện cách nhau 50ms
    useEffect(() => {
        items.forEach((item, index) => {
            setTimeout(() => {
                setVisibleItems(prev => new Set([...prev, item.id]));
            }, index * 50);
        });
    }, [items]);

    const handleRemove = (id: string) => {
        setRemovingId(id);
        // Đợi animation exit xong mới xóa thật
        setTimeout(() => {
            // Call API delete + remove from list
            setRemovingId(null);
        }, 300);
    };

    const handleResume = (item: HistoryItem) => {
        navigate(`/book/${item.publication.id}/read`, {
            state: { 
                isComic: item.publication.formatType === 2,
                resumeChapter: item.chapterNumber,
                resumePercent: item.scrollPercent
            }
        });
    };

    return (
        <div style={{ display: 'flex', flexDirection: 'column', gap: '12px' }}>
            {items.map(item => (
                <div
                    key={item.id}
                    style={{
                        display: 'flex', gap: '16px', padding: '16px',
                        backgroundColor: '#fff', borderRadius: '12px',
                        boxShadow: '0 1px 3px rgba(0,0,0,0.05)',
                        // Animation states:
                        opacity: removingId === item.id ? 0 : visibleItems.has(item.id) ? 1 : 0,
                        transform: removingId === item.id
                            ? 'translateX(100px) scale(0.95)'
                            : visibleItems.has(item.id) ? 'translateY(0)' : 'translateY(20px)',
                        transition: 'all 300ms cubic-bezier(0.4, 0, 0.2, 1)',
                        maxHeight: removingId === item.id ? '0px' : '200px',
                        overflow: 'hidden',
                        marginBottom: removingId === item.id ? '-12px' : '0',
                    }}
                >
                    {/* Cover */}
                    <img src={item.publication.coverImageUrl} alt="" 
                         style={{ width: 60, height: 80, borderRadius: 8, objectFit: 'cover' }} />
                    
                    {/* Info */}
                    <div style={{ flex: 1 }}>
                        <h4 style={{ fontSize: '15px', fontWeight: 600, marginBottom: 4 }}>
                            {item.publication.title}
                        </h4>
                        <p style={{ fontSize: '13px', color: '#6b7280', marginBottom: 8 }}>
                            Chương {item.chapterNumber}: {item.chapterTitle}
                        </p>
                        {/* Progress mini bar */}
                        <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
                            <div style={{ flex: 1, height: 4, backgroundColor: '#e5e7eb', borderRadius: 2 }}>
                                <div style={{ width: `${item.scrollPercent}%`, height: '100%', backgroundColor: item.scrollPercent === 100 ? '#22c55e' : '#2196f3', borderRadius: 2, transition: 'width 500ms ease' }} />
                            </div>
                            <span style={{ fontSize: '12px', color: '#6b7280', minWidth: 35 }}>
                                {item.scrollPercent === 100 ? '✅' : `${Math.round(item.scrollPercent)}%`}
                            </span>
                        </div>
                    </div>

                    {/* Actions */}
                    <div style={{ display: 'flex', flexDirection: 'column', gap: 6 }}>
                        <button onClick={() => handleResume(item)}
                            style={{ padding: '6px 14px', backgroundColor: '#2196f3', color: '#fff', border: 'none', borderRadius: 6, fontSize: '12px', fontWeight: 600, cursor: 'pointer' }}>
                            ▶️ Tiếp tục
                        </button>
                        <button onClick={() => handleRemove(item.id)}
                            style={{ padding: '4px 14px', backgroundColor: 'transparent', color: '#9ca3af', border: '1px solid #e5e7eb', borderRadius: 6, fontSize: '11px', cursor: 'pointer' }}>
                            Xóa
                        </button>
                    </div>
                </div>
            ))}
        </div>
    );
};

6.1 Hook: useScrollReveal

📁 xom-truyen/src/hooks/useScrollReveal.ts
import { useRef, useEffect, useState } from 'react';

/**
 * Hook cho scroll-triggered reveal animation.
 * Element sẽ animate vào khi xuất hiện trong viewport.
 */
export function useScrollReveal(options?: { threshold?: number; rootMargin?: string; delay?: number }) {
    const ref = useRef<HTMLDivElement>(null);
    const [isVisible, setIsVisible] = useState(false);

    useEffect(() => {
        const observer = new IntersectionObserver(
            ([entry]) => {
                if (entry.isIntersecting) {
                    if (options?.delay) {
                        setTimeout(() => setIsVisible(true), options.delay);
                    } else {
                        setIsVisible(true);
                    }
                    observer.disconnect(); // Chỉ reveal 1 lần
                }
            },
            { threshold: options?.threshold || 0.1, rootMargin: options?.rootMargin || '0px' }
        );

        if (ref.current) observer.observe(ref.current);
        return () => observer.disconnect();
    }, []);

    const style: React.CSSProperties = {
        opacity: isVisible ? 1 : 0,
        transform: isVisible ? 'translateY(0)' : 'translateY(30px)',
        transition: 'opacity 500ms ease, transform 500ms cubic-bezier(0.4, 0, 0.2, 1)',
    };

    return { ref, isVisible, style };
}

 * - Ban đầu hiện 10 chương
 * - "Xem thêm" → expand smooth (max-height transition)
 * - Highlight chương đang đọc (pulse animation)
 * - Locked chapters có icon 🔒 + opacity thấp hơn
 */
export const AnimatedChapterList: React.FC<{ chapters: Chapter[]; currentChapter?: number }> = 
    ({ chapters, currentChapter }) => {
    const [expanded, setExpanded] = useState(false);
    const displayChapters = expanded ? chapters : chapters.slice(0, 10);

    return (
        <div>
            <h3 style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
                Danh sách chương ({chapters.length})
                <button onClick={() => setExpanded(!expanded)}
                    style={{ fontSize: '14px', color: '#2196f3', background: 'none', border: 'none', cursor: 'pointer' }}>
                    {expanded ? 'Thu gọn ▲' : `Xem tất cả ${chapters.length} chương ▼`}
                </button>
            </h3>

            <div style={{
                maxHeight: expanded ? `${chapters.length * 52}px` : '520px',
                overflow: 'hidden',
                transition: 'max-height 500ms cubic-bezier(0.4, 0, 0.2, 1)',
            }}>
                {displayChapters.map((ch, idx) => (
                    <div key={ch.id} style={{
                        display: 'flex', justifyContent: 'space-between',
                        padding: '12px 16px', borderBottom: '1px solid #f3f4f6',
                        backgroundColor: ch.chapterNumber === currentChapter ? '#eff6ff' : 'transparent',
                        transition: 'background-color 200ms ease',
                        opacity: ch.isLocked ? 0.6 : 1,
                        // Stagger enter animation:
                        animation: `slideInLeft 300ms ${idx * 30}ms both`,
                    }}>
                        <span>{ch.isLocked ? '🔒 ' : ''} Chương {ch.chapterNumber}: {ch.title}</span>
                        <span style={{ fontSize: '12px', color: '#9ca3af' }}>{ch.updatedAt}</span>
                    </div>
                ))}
            </div>
        </div>
    );
};

9.2 Nguyên tắc performance

  • Chỉ animate opacity + transform: 2 property này được GPU accelerate, không gây layout/paint
  • will-change: Thêm will-change: transform, opacity cho elements sẽ animate
  • IntersectionObserver: Không dùng scroll event listener (gây jank) → dùng IO cho reveal
  • requestAnimationFrame: Dùng rAF cho continuous animations (progress bar, auto-scroll)
  • prefers-reduced-motion: Tắt animations cho users nhạy cảm
  • Debounce scroll events: Nếu phải dùng scroll → debounce 16ms (1 frame)
💡 Thư viện tùy chọn: Nếu muốn animations phức tạp hơn (gesture, spring physics, layout animations), có thể thêm framer-motion (55KB gzipped). Nhưng với danh sách trên, CSS + inline styles là đủ, không cần thêm dependency.
