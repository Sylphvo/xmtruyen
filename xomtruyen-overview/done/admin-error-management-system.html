3. React Error Boundary (Frontend Crash Handler)

Vấn đề: Nếu bất kỳ component nào throw error trong render → toàn bộ React app crash → trang trắng. Error Boundary bắt crash và hiển thị fallback UI thân thiện.
📁 xomtruyen-admin/src/components/ErrorBoundary.tsx
import React from 'react';
import { errorService } from '../services/errorService';

interface State {
    hasError: boolean;
    error: Error | null;
    errorInfo: React.ErrorInfo | null;
}

/**
 * Global Error Boundary — bọc toàn bộ App.
 * Bắt mọi unhandled error trong component tree → hiện fallback UI.
 * Ghi log lỗi vào errorService.
 */
export class ErrorBoundary extends React.Component<{ children: React.ReactNode }, State> {
    state: State = { hasError: false, error: null, errorInfo: null };

    static getDerivedStateFromError(error: Error): Partial<State> {
        return { hasError: true, error };
    }

    componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
        // Ghi log vào hệ thống Error Management
        errorService.captureError({
            type: 'REACT_CRASH',
            severity: 'critical',
            message: error.message,
            stack: error.stack || '',
            componentStack: errorInfo.componentStack || '',
            url: window.location.href,
            timestamp: new Date().toISOString(),
            userAgent: navigator.userAgent
        });

        this.setState({ errorInfo });
    }

    handleReset = () => {
        this.setState({ hasError: false, error: null, errorInfo: null });
    };

    render() {
        if (this.state.hasError) {
            return (
                <div style={{
                    display: 'flex', flexDirection: 'column', alignItems: 'center',
                    justifyContent: 'center', minHeight: '100vh', padding: '40px',
                    backgroundColor: '#f8fafc', textAlign: 'center'
                }}>
                    <div style={{ fontSize: '48px', marginBottom: '16px' }}>💥</div>
                    <h2 style={{ color: '#1f2937', marginBottom: '8px' }}>Ứng dụng gặp sự cố</h2>
                    <p style={{ color: '#6b7280', marginBottom: '24px', maxWidth: '400px' }}>
                        Đã xảy ra lỗi không mong muốn. Lỗi đã được ghi nhận và đội ngũ kỹ thuật sẽ khắc phục sớm nhất.
                    </p>

                    {/* Error details (dev mode) */}
                    {process.env.NODE_ENV === 'development' && this.state.error && (
                        <details style={{ marginBottom: '20px', textAlign: 'left', maxWidth: '600px' }}>
                            <summary style={{ cursor: 'pointer', color: '#ef4444' }}>Chi tiết lỗi (Dev)</summary>
                            <pre style={{ fontSize: '12px', overflow: 'auto', padding: '12px', background: '#1f2937', color: '#f87171', borderRadius: '8px', marginTop: '8px' }}>
                                {this.state.error.message}\n\n{this.state.error.stack}
                            </pre>
                        </details>
                    )}

                    <div style={{ display: 'flex', gap: '12px' }}>
                        <button onClick={this.handleReset} style={{
                            padding: '10px 24px', backgroundColor: '#2196f3', color: '#fff',
                            border: 'none', borderRadius: '8px', fontWeight: 600, cursor: 'pointer'
                        }}>
                            Thử lại
                        </button>
                        <button onClick={() => window.location.href = '/'} style={{
                            padding: '10px 24px', backgroundColor: '#e5e7eb', color: '#374151',
                            border: 'none', borderRadius: '8px', fontWeight: 600, cursor: 'pointer'
                        }}>
                            Về trang chủ
                        </button>
                    </div>
                </div>
            );
        }
        return this.props.children;
    }
}

4. Centralized Error Service (Thu thập & Phân loại)

📁 xomtruyen-admin/src/services/errorService.ts
/**
 * Centralized Error Service — Thu thập, phân loại và gửi lỗi lên server.
 * Thay thế console.error + toast rời rạc bằng 1 hệ thống tập trung.
 */

export type ErrorSeverity = 'critical' | 'high' | 'medium' | 'low';
export type ErrorCategory = 'REACT_CRASH' | 'API_ERROR' | 'NETWORK' | 'AUTH' | 'VALIDATION' | 'UNKNOWN';

export interface AppError {
    id?: string;
    type: ErrorCategory;
    severity: ErrorSeverity;
    message: string;
    stack?: string;
    componentStack?: string;
    url: string;
    endpoint?: string;
    statusCode?: number;
    requestBody?: string;
    responseBody?: string;
    timestamp: string;
    userAgent: string;
    userId?: string;
    metadata?: Record<string, any>;
}

class ErrorService {
    private queue: AppError[] = [];
    private maxQueueSize = 100;
    private flushInterval: number;
    private listeners: ((error: AppError) => void)[] = [];

    constructor() {
        // Flush queue mỗi 30 giây
        this.flushInterval = window.setInterval(() => this.flush(), 30000);

        // Bắt unhandled promise rejections
        window.addEventListener('unhandledrejection', (event) => {
            this.captureError({
                type: 'UNKNOWN',
                severity: 'high',
                message: `Unhandled Promise Rejection: ${event.reason?.message || event.reason}`,
                stack: event.reason?.stack || '',
                url: window.location.href,
                timestamp: new Date().toISOString(),
                userAgent: navigator.userAgent
            });
        });

        // Bắt global JS errors
        window.addEventListener('error', (event) => {
            this.captureError({
                type: 'UNKNOWN',
                severity: 'high',
                message: event.message,
                stack: event.error?.stack || `${event.filename}:${event.lineno}:${event.colno}`,
                url: window.location.href,
                timestamp: new Date().toISOString(),
                userAgent: navigator.userAgent
            });
        });
    }

    /**
     * Ghi nhận 1 lỗi — phân loại + lưu queue + notify listeners
     */
    captureError(error: AppError) {
        // Assign ID
        error.id = `err_${Date.now()}_${Math.random().toString(36).substr(2, 5)}`;

        // Thêm userId nếu đã login
        const token = sessionStorage.getItem('accessToken');
        if (token) {
            try {
                const payload = JSON.parse(atob(token.split('.')[1]));
                error.userId = payload.sub || payload.nameid;
            } catch (e) { /* ignore */ }
        }

        // Auto-detect severity nếu chưa set
        if (!error.severity) {
            error.severity = this.detectSeverity(error);
        }

        // Thêm vào queue
        this.queue.push(error);
        if (this.queue.length > this.maxQueueSize) {
            this.queue.shift(); // Bỏ cũ nhất
        }

        // Lưu localStorage (offline backup)
        this.saveToLocalStorage(error);

        // Notify listeners (UI updates)
        this.listeners.forEach(fn => fn(error));

        // Nếu critical → flush ngay
        if (error.severity === 'critical') {
            this.flush();
        }

        // Dev mode: vẫn console.error
        if (process.env.NODE_ENV === 'development') {
            console.error(`[ErrorService] [${error.severity}] ${error.type}:`, error.message);
        }
    }

    /**
     * Tạo error từ Axios response
     */
    captureApiError(axiosError: any, endpoint?: string) {
        const status = axiosError.response?.status;
        const category: ErrorCategory = 
            !axiosError.response ? 'NETWORK' :
            status === 401 || status === 403 ? 'AUTH' :
            status === 400 || status === 422 ? 'VALIDATION' :
            'API_ERROR';

        this.captureError({
            type: category,
            severity: status >= 500 ? 'high' : status === 401 ? 'medium' : 'low',
            message: axiosError.response?.data?.message || axiosError.message || 'API Error',
            endpoint: endpoint || axiosError.config?.url,
            statusCode: status,
            requestBody: JSON.stringify(axiosError.config?.data)?.substring(0, 500),
            responseBody: JSON.stringify(axiosError.response?.data)?.substring(0, 500),
            url: window.location.href,
            timestamp: new Date().toISOString(),
            userAgent: navigator.userAgent
        });
    }

    /**
     * Gửi queue lên backend
     */
    private async flush() {
        if (this.queue.length === 0) return;

        const errors = [...this.queue];
        this.queue = [];

        try {
            await fetch('/api/admin/error-logs/batch', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'Authorization': `Bearer ${sessionStorage.getItem('accessToken') || ''}`
                },
                body: JSON.stringify({ errors }),
                keepalive: true
            });
        } catch (e) {
            // Nếu gửi thất bại → đưa lại vào queue
            this.queue.unshift(...errors);
        }
    }

    private saveToLocalStorage(error: AppError) {
        try {
            const existing = JSON.parse(localStorage.getItem('xomtruyen_error_log') || '[]');
            existing.push(error);
            if (existing.length > 50) existing.splice(0, existing.length - 50);
            localStorage.setItem('xomtruyen_error_log', JSON.stringify(existing));
        } catch (e) { /* quota exceeded — ignore */ }
    }

    private detectSeverity(error: AppError): ErrorSeverity {
        if (error.type === 'REACT_CRASH') return 'critical';
        if (error.statusCode && error.statusCode >= 500) return 'high';
        if (error.type === 'NETWORK') return 'medium';
        return 'low';
    }

    /** Subscribe to new errors (for UI notification badge) */
    onError(listener: (error: AppError) => void) {
        this.listeners.push(listener);
        return () => { this.listeners = this.listeners.filter(l => l !== listener); };
    }

    /** Get recent errors from localStorage (for offline view) */
    getRecentErrors(): AppError[] {
        try {
            return JSON.parse(localStorage.getItem('xomtruyen_error_log') || '[]');
        } catch { return []; }
    }

    /** Clear local cache */
    clearLocalErrors() {
        localStorage.removeItem('xomtruyen_error_log');
    }

    destroy() {
        clearInterval(this.flushInterval);
    }
}

export const errorService = new ErrorService();

            metadata: new { freeGb, usedPercent = 100 - (int)(drive.AvailableFreeSpace * 100 / drive.TotalSize) }));

        result.OverallStatus = result.Services.All(s => s.Status == "UP") ? "HEALTHY" :
                              result.Services.Any(s => s.Status == "DOWN") ? "UNHEALTHY" : "DEGRADED";

        return Ok(result);

/// <summary>
/// Middleware bắt MỌI exception không xử lý trong pipeline ASP.NET.
/// Ghi log vào DB + trả response chuẩn cho client.
/// </summary>
public class GlobalExceptionMiddleware
{
    private readonly RequestDelegate _next;
    private readonly ILogger<GlobalExceptionMiddleware> _logger;

    public async Task InvokeAsync(HttpContext context, ApplicationDbContext db)
    {
        try
        {
            await _next(context);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Unhandled exception: {Path}", context.Request.Path);

            // Ghi vào DB
            var errorLog = new ErrorLog
            {
                Id = Guid.NewGuid(),
                Severity = "high",
                Category = "SERVER_EXCEPTION",
                Message = ex.Message,
                StackTrace = ex.StackTrace ?? "",
                Endpoint = $"{context.Request.Method} {context.Request.Path}",
                StatusCode = 500,
                UserId = context.User?.FindFirst("sub")?.Value,
                IpAddress = context.Connection.RemoteIpAddress?.ToString(),
                UserAgent = context.Request.Headers["User-Agent"].ToString(),
                CreatedAt = DateTime.UtcNow
            };
            db.ErrorLogs.Add(errorLog);
            await db.SaveChangesAsync();

            // Trả response chuẩn (KHÔNG lộ stack trace ra production)
            context.Response.StatusCode = 500;
            context.Response.ContentType = "application/json";
            await context.Response.WriteAsJsonAsync(new
            {
                success = false,
                message = "Đã xảy ra lỗi hệ thống. Vui lòng thử lại sau.",
                errorId = errorLog.Id, // Để user report kèm ID
                timestamp = errorLog.CreatedAt
            });
        }
    }
}

8. Auto-Retry & Circuit Breaker (Frontend)

📁 xomtruyen-admin/src/utils/retryFetch.ts
import { errorService } from '../services/errorService';

interface RetryOptions {
    maxRetries?: number;       // Số lần retry tối đa (default: 3)
    retryDelay?: number;       // Delay giữa các lần retry (ms, default: 1000)
    retryOn?: number[];        // Status codes nào sẽ retry (default: [408, 500, 502, 503, 504])
    onRetry?: (attempt: number, error: any) => void;
}

/**
 * Wrapper retry logic cho API calls.
 * Tự động retry khi gặp lỗi tạm thời (network, 5xx).
 * Không retry cho lỗi client (4xx) vì retry cũng không fix được.
 */
export async function withRetry<T>(
    fn: () => Promise<T>,
    options: RetryOptions = {}
): Promise<T> {
    const { maxRetries = 3, retryDelay = 1000, retryOn = [408, 500, 502, 503, 504], onRetry } = options;

    let lastError: any;

    for (let attempt = 1; attempt <= maxRetries + 1; attempt++) {
        try {
            return await fn();
        } catch (error: any) {
            lastError = error;
            const status = error.response?.status;

            // Không retry nếu là lỗi client (4xx) trừ 408 timeout
            if (status && status >= 400 && status < 500 && !retryOn.includes(status)) {
                throw error;
            }

            // Đã hết lượt retry
            if (attempt > maxRetries) {
                errorService.captureApiError(error);
                throw error;
            }

            // Callback thông báo đang retry
            onRetry?.(attempt, error);

            // Exponential backoff: 1s, 2s, 4s...
            const delay = retryDelay * Math.pow(2, attempt - 1);
            await new Promise(resolve => setTimeout(resolve, delay));
        }
    }

    throw lastError;
}

12.3 Resolution Service (Backend)

📁 xomtruyen.API/Services/ErrorResolutionService.cs
/// <summary>
/// Service tìm giải pháp khắc phục cho lỗi dựa trên Knowledge Base.
/// Pattern matching: message, category, endpoint, status code.
/// </summary>
public class ErrorResolutionService
{
    private readonly ApplicationDbContext _db;

    /// <summary>
    /// Tìm giải pháp đề xuất cho 1 error log
    /// </summary>
    public async Task<List<ResolutionSuggestion>> GetSuggestionsAsync(ErrorLog error)
    {
        var suggestions = new List<ResolutionSuggestion>();

        // 1. Exact match: cùng endpoint + status code
        var exactMatches = await _db.ErrorResolutions
            .Where(r => r.IsActive)
            .Where(r => r.EndpointMatch != null && error.Endpoint != null 
                        && error.Endpoint.Contains(r.EndpointMatch))
            .Where(r => r.StatusCodeMatch == null || r.StatusCodeMatch == error.StatusCode)
            .OrderByDescending(r => r.TimesApplied)
            .Take(3)
            .ToListAsync();

        foreach (var match in exactMatches)
        {
            suggestions.Add(new ResolutionSuggestion
            {
                Resolution = match,
                MatchType = "exact",
                Confidence = 0.9
            });
        }

        // 2. Pattern match: message contains pattern
        if (suggestions.Count < 3)
        {
            var patternMatches = await _db.ErrorResolutions
                .Where(r => r.IsActive)
                .Where(r => r.CategoryMatch == error.Category || r.CategoryMatch == null)
                .Where(r => EF.Functions.ILike(error.Message, "%" + r.ErrorPattern + "%"))
                .OrderByDescending(r => r.SuccessRate)
                .Take(3 - suggestions.Count)
                .ToListAsync();

            foreach (var match in patternMatches)
            {
                suggestions.Add(new ResolutionSuggestion
                {
                    Resolution = match,
                    MatchType = "pattern",
                    Confidence = 0.7
                });
            }
        }

        // 3. Fallback: dựa trên category + severity (generic suggestions)
        if (suggestions.Count == 0)
        {
            suggestions.AddRange(GetGenericSuggestions(error));
        }

        return suggestions;
    }

    /// <summary>
    /// Gợi ý chung khi không tìm thấy match trong KB
    /// </summary>
    private List<ResolutionSuggestion> GetGenericSuggestions(ErrorLog error)
    {
        return error.Category switch
        {
            "NETWORK" => new List<ResolutionSuggestion>
            {
                new() { GenericTitle = "Kiểm tra kết nối mạng", 
                         GenericDescription = "• Kiểm tra server API có đang chạy\n• Kiểm tra firewall không chặn port\n• Thử ping server: curl http://localhost:5000/api/system/health",
                         MatchType = "generic", Confidence = 0.3 }
            },
            "AUTH" => new List<ResolutionSuggestion>
            {
                new() { GenericTitle = "Lỗi xác thực / phân quyền",
                         GenericDescription = "• Token có thể đã hết hạn → đăng nhập lại\n• Kiểm tra role user có đủ quyền truy cập endpoint\n• Kiểm tra JWT Secret trong appsettings.json",
                         MatchType = "generic", Confidence = 0.3 }
            },
            "API_ERROR" when error.StatusCode == 500 => new List<ResolutionSuggestion>
            {
                new() { GenericTitle = "Lỗi server 500 — Internal Server Error",
                         GenericDescription = "• Kiểm tra log backend (.NET): dotnet run → xem console\n• Kiểm tra connection string DB\n• Kiểm tra EF migration đã update chưa: dotnet ef database update\n• Xem chi tiết exception trong ErrorLogs table",
                         MatchType = "generic", Confidence = 0.4 }
            },
            "VALIDATION" => new List<ResolutionSuggestion>
            {
                new() { GenericTitle = "Lỗi validation dữ liệu",
                         GenericDescription = "• Kiểm tra các trường bắt buộc đã điền đủ chưa\n• Kiểm tra format dữ liệu (email, số, ngày tháng)\n• Xem response body chi tiết để biết trường nào lỗi",
                         MatchType = "generic", Confidence = 0.5 }
            },
            _ => new List<ResolutionSuggestion>
            {
                new() { GenericTitle = "Lỗi chưa xác định",
                         GenericDescription = "• Kiểm tra console browser (F12) để xem chi tiết\n• Kiểm tra network tab xem request/response\n• Thử refresh trang và thao tác lại\n• Nếu lặp lại → ghi chú lại steps to reproduce",
                         MatchType = "generic", Confidence = 0.2 }
            }
        };
    }

    /// <summary>
    /// Áp dụng giải pháp: cập nhật stats + liên kết error↔resolution
    /// </summary>
    public async Task ApplyResolutionAsync(Guid errorLogId, Guid resolutionId, Guid adminId, bool wasSuccessful, string? notes)
    {
        // Liên kết
        _db.Set<ErrorLogResolution>().Add(new ErrorLogResolution
        {
            ErrorLogId = errorLogId,
            ResolutionId = resolutionId,
            AppliedBy = adminId,
            WasSuccessful = wasSuccessful,
            Notes = notes
        });

        // Cập nhật stats resolution
        var resolution = await _db.ErrorResolutions.FindAsync(resolutionId);
        if (resolution != null)
        {
            resolution.TimesApplied++;
            resolution.LastAppliedAt = DateTime.UtcNow;
            // Tính success rate
            var total = await _db.Set<ErrorLogResolution>().CountAsync(r => r.ResolutionId == resolutionId);
            var success = await _db.Set<ErrorLogResolution>().CountAsync(r => r.ResolutionId == resolutionId && r.WasSuccessful);
            resolution.SuccessRate = total > 0 ? (double)success / total : 0;
        }

        // Mark error là resolved
        var errorLog = await _db.ErrorLogs.FindAsync(errorLogId);
        if (errorLog != null)
        {
            errorLog.IsResolved = true;
            errorLog.ResolvedBy = adminId;
            errorLog.ResolvedAt = DateTime.UtcNow;
            errorLog.ResolvedNote = notes;
        }

        await _db.SaveChangesAsync();
    }
}

13.2 API Integration — Quy tắc gọi API an toàn

Checklist cho MỖI API call trong admin:

  1. Wrap trong try/catch — KHÔNG BAO GIỜ để promise unhandled
  2. Loading state — Hiện skeleton/spinner khi đang fetch, disable buttons khi đang submit
  3. Error state — Hiện message rõ ràng khi API fail (toast + log vào errorService)
  4. Empty state — Hiện UI "Không có dữ liệu" khi API trả mảng rỗng (không blank trắng)
  5. Timeout handling — Set timeout cho mọi request (60s default, 300s cho upload)
  6. Cancel on unmount — Dùng AbortController để cancel request khi user navigate away
  7. Optimistic rollback — Nếu optimistic update fail → rollback state về trước
� Pattern mẫu cho API call an toàn
// ✅ PATTERN CHUẨN — Mọi page nên theo pattern này:

const [data, setData] = useState<IBook[]>([]);       // Default rỗng, KHÔNG undefined
const [isLoading, setIsLoading] = useState(true);      // Loading ngay từ đầu
const [error, setError] = useState<string | null>(null);

useEffect(() => {
    const controller = new AbortController();           // Cancel on unmount
    
    const fetchData = async () => {
        setIsLoading(true);
        setError(null);
        try {
            const res = await getBooks({ page: 1 }, { signal: controller.signal });
            setData(res?.data ?? []);                   // Null-safe + default
        } catch (err: any) {
            if (err.name === 'CanceledError') return;  // Bỏ qua nếu canceled
            setError(err.message || 'Có lỗi xảy ra');
            errorService.captureApiError(err);         // Ghi log tập trung
        } finally {
            setIsLoading(false);
        }
    };
    
    fetchData();
    return () => controller.abort();                    // Cleanup
}, []);

// Trong JSX:
{isLoading ? <TableSkeleton /> :

13.7 Backend — Ngăn lỗi từ API layer

Checklist cho mỗi API endpoint (.NET):

  1. Input validation — [Required], [MaxLength], [Range] trên mọi DTO property
  2. ModelState check — if (!ModelState.IsValid) return BadRequest(ModelState)
  3. Null check entity — var book = await _db.Books.FindAsync(id); if (book == null) return NotFound();
  4. Try-catch cho external calls — DB, file system, external API → wrap trong try-catch
  5. Trả response chuẩn — Luôn return ApiResponse<T> với success/message/data — KHÔNG throw raw exception ra client
  6. Log meaningful info — Log đủ context: userId, endpoint, params, error. KHÔNG log sensitive data (password, token)
  7. Idempotent operations — PUT/DELETE phải idempotent (gọi nhiều lần = kết quả như gọi 1 lần)
  8. Pagination protection — Giới hạn pageSize tối đa (≤100) — ngăn client request 1 triệu rows
// ✅ Controller endpoint chuẩn (phòng lỗi tối đa):

[HttpPut("{id}")]
[Authorize(Roles = "Admin,Editor")]
public async Task<IActionResult> UpdateBook(Guid id, [FromBody] UpdateBookRequest request)
{
    // 1. Validate input
    if (!ModelState.IsValid)
        return BadRequest(ApiResponse.Error(ModelState.GetErrors()));

    // 2. Check entity exists
    var book = await _db.Publications.FindAsync(id);
    if (book == null)
        return NotFound(ApiResponse.Error($"Sách với ID {id} không tồn tại"));

    // 3. Business logic (wrapped)
    try
    {
        book.Title = request.Title.Trim();      // Trim input
        book.Author = request.Author.Trim();
        book.UpdatedAt = DateTime.UtcNow;
        book.UpdatedBy = GetCurrentUserId();
        
        await _db.SaveChangesAsync();
        
        _logger.LogInformation("Book {BookId} updated by {UserId}", id, GetCurrentUserId());
        return Ok(ApiResponse.Ok("Cập nhật thành công"));
    }
    catch (DbUpdateException ex)
    {
        _logger.LogError(ex, "DB error updating book {BookId}", id);
        return StatusCode(500, ApiResponse.Error("Lỗi cập nhật database"));
    }
}
